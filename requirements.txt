# =============================================================================
# luminous - Python dependencies (conda-friendly list)
# =============================================================================
#
# One package per line. Comments are kept on their own lines so that conda's
# `--file` parser (which is stricter than pip's) can consume this verbatim.
#
# Install into a conda env (you run these yourself):
#
#   conda create -n luminous python=3.10
#   conda activate luminous
#   conda config --env --add channels conda-forge
#   conda install --file requirements.txt
#
# `conda-forge` is needed for vtk (and is the usual source for pillow /
# matplotlib builds on osx-arm64).
#
# Equivalent pip form, if you prefer pip inside the env:
#   pip install -r requirements.txt
#
# -----------------------------------------------------------------------------
# Python version
# -----------------------------------------------------------------------------
# pyproject.toml declares no `requires-python`. The code relies on
# `from __future__ import annotations` plus PEP 604 (`X | None`) annotations,
# so Python >= 3.10 is required. Verified against conda env "luminous"
# (Python 3.10.22); test caches also exist for 3.11.
# `python` itself is intentionally NOT listed below so the conda solver is
# free to resolve the interpreter you already pinned in the env.
#
# -----------------------------------------------------------------------------
# Source of truth
# -----------------------------------------------------------------------------
# Every third-party import reachable from luminous/src/** and from the example
# scripts in the parent repo (luminous-examples). NOTE: pyproject.toml
# currently declares `dependencies = []`, so this file is the only place these
# runtime requirements are documented.
# =============================================================================


# -----------------------------------------------------------------------------
# Runtime - required
# -----------------------------------------------------------------------------

# Core numeric layer: vector math, ray/shape intersection, detector arrays.
# `from numpy.typing import NDArray` (detector.py, scene.py) requires >= 1.21.
#
# PINNED <2 ON PURPOSE. The code still uses three aliases that NumPy 2.0
# removed, so NumPy 2.x breaks it:
#   src/radiometry/luminous.py:121,152  np.Inf   (default arg -> raises at IMPORT)
#   src/radiometry/luminous.py:82       np.float_
#   src/radiometry/luminous.py:262      np.trapz (renamed np.trapezoid)
# conda-forge's newest numpy is 2.x, so do NOT unpin this until those three
# call sites are migrated (np.inf, np.float64, np.trapezoid).
numpy<2


# Peak finding, peak widths and FWHM: scipy.signal.find_peaks / peak_widths,
# imported lazily inside Spectrum.peak_amplitudes / peak_indices /
# full_width_half_max (src/radiometry/luminous.py).
scipy

# Detector image assembly: top-level `from PIL import Image` in
# src/element/detector.py (Image.fromarray / Image.merge). Imported at module
# scope, so the library cannot be imported without it. Conda/PyPI name is
# `pillow`, not `PIL`.
pillow

# Plotting for every example script in the parent repo (plt.imshow, plot,
# savefig, show) and for the Spectrum plot helper in src/radiometry/luminous.py.
matplotlib


# -----------------------------------------------------------------------------
# Development / test - required to run the test suite
# -----------------------------------------------------------------------------

# Test runner for luminous/tests/** (pytest.ini markers, ./run_tests.sh).
pytest

# Provides --cov / --cov-report used by `./run_tests.sh --coverage`, and the
# --no-cov flag referenced by .vscode/launch.json.
pytest-cov

# Build backend declared in pyproject.toml ([build-system] requires).
setuptools


# -----------------------------------------------------------------------------
# Optional - uncomment only if you use the 3D ray debugger
# -----------------------------------------------------------------------------
# vtk is imported lazily inside ConcreteRayDebugger.__init__
# (src/utilities/ray_debugger.py). NullRayDebugger (the default) needs no vtk,
# so the library runs fine without it. Needed only for
# Scene.attach_ray_debugger(...) and the attach_ray_debugger*.py examples.
# Large download; install from conda-forge.
vtk
